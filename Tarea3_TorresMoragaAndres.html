# Filtrar todos los montos del canal completo
montos.canal <- transacciones %>%

  pull(monto_crc)

ej2.a <- round(benford.mad(montos.canal), 4)


ej2.b <- ej2%>%
  slice_max(mad, n = 1) %>% 
  select(canton)


ej2.c <- ej2 %>% 
  filter(mad > 0.015) %>% 
  nrow()


ej2.d <- benford.canton(transacciones, "Web", cantones) %>%
  slice_max(mad, n = 1) %>% 
  select(canton)


# 1. Preparar datos de Benford teóricos y observados
benford.teorico <- data.frame(
  digito = factor(1:9),
  p.benford = log10(1 + 1 / (1:9))
)

datos.grafico <- transacciones %>%
  filter(canal == canal.benford, !is.na(monto_crc), monto_crc > 0) %>%
  mutate(digito = factor(as.numeric(substr(as.character(monto_crc), 1, 1)))) %>%
  filter(digito %in% factor(1:9)) %>%
  count(digito) %>%
  mutate(p.d = n / sum(n)) %>%
  right_join(benford.teorico, by = "digito") %>%
  mutate(p.d = coalesce(p.d, 0))

# 2. Generar el gráfico de barras vs curva/línea de Benford
figura.ej2 <- ggplot(datos.grafico, aes(x = digito)) +
  geom_col(aes(y = p.d, fill = "Canal (Observado)"), alpha = 0.7, width = 0.6) +
  geom_line(aes(y = p.benford, group = 1, color = "Ley de Benford"), size = 1.2) +
  geom_point(aes(y = p.benford, color = "Ley de Benford"), size = 2.5) +
  scale_fill_manual(name = "", values = c("Canal (Observado)" = "#2b5c8f")) +
  scale_color_manual(name = "", values = c("Ley de Benford" = "#d95f02")) +
  scale_y_continuous(labels = scales::percent_format(accuracy = 1)) +
  labs(
    title = paste("Distribución del Primer Dígito - Canal", canal.benford),
    x = "Primer Dígito (d)",
    y = "Proporción (p.d)"
  ) +
  theme_minimal() +
  theme(legend.position = "top")


ggsave(
  filename = "info/figuras/ley.Benford.ej2.png", 
  plot = figura.ej2, 
  width = 10, 
  height = 5, 
  dpi = 300
)
print(figura.ej2)
